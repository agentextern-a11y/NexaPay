export async function loadTest(concurrency: number = 10, iterations: number = 100) {
  const results: { success: number; failure: number; latencies: number[] } = {
    success: 0,
    failure: 0,
    latencies: [],
  };
  
  const tasks = Array(iterations).fill(null).map(async () => {
    const start = Date.now();
    try {
      const res = await fetch(`${API_BASE}/compliance/assess`, {
        method: 'POST',
        body: JSON.stringify({
          userId: '00000000-0000-0000-0000-000000000000',
          cardId: '00000000-0000-0000-0000-000000000000',
          amount: Math.floor(Math.random() * 10000) + 100,
          merchantName: 'Load Test Merchant',
          merchantCountry: 'US',
          merchantCategory: '5411',
          isNfc: Math.random() > 0.5,
        }),
        headers: { 'Content-Type': 'application/json' },
      });
      
      const latency = Date.now() - start;
      results.latencies.push(latency);
      
      if (res.ok) {
        results.success++;
      } else {
        results.failure++;
      }
    } catch {
      results.failure++;
    }
  });
  
  // Run in batches for controlled concurrency
  for (let i = 0; i < tasks.length; i += concurrency) {
    const batch = tasks.slice(i, i + concurrency);
    await Promise.all(batch);
  }
  
  return {
    ...results,
    avgLatency: results.latencies.reduce((a, b) => a + b, 0) / results.latencies.length,
    p95Latency: results.latencies.sort((a, b) => a - b)[Math.floor(results.latencies.length * 0.95)],
    p99Latency: results.latencies.sort((a, b) => a - b)[Math.floor(results.latencies.length * 0.99)],
  };
}

import { describe, it, expect } from 'bun:test';
import { db } from '@nexa-pay/db';
import { ledgerAccounts, ledgerEntries } from '@nexa-pay/db/schema';
import { eq, sql } from 'drizzle-orm';

describe('Ledger System', () => {
  
  const testUserId = '00000000-0000-0000-0000-000000000000';
  let accountId: string;
  
  // Clean up before tests
  beforeAll(async () => {
    await db.delete(ledgerEntries);
    await db.delete(ledgerAccounts);
  });
  
  it('should create a ledger account', async () => {
    const [account] = await db.insert(ledgerAccounts)
      .values({
        userId: testUserId,
        accountType: 'USER_WALLET',
        currency: 'USDC',
        balance: '1000.00',
      })
      .returning();
    
    expect(account).toBeDefined();
    expect(account!.balance).toBe('1000.00');
    accountId = account!.id;
  });
  
  it('should record a debit entry', async () => {
    const balanceBefore = 1000.00;
    const debitAmount = 50.00;
    const balanceAfter = balanceBefore - debitAmount;
    
    await db.transaction(async (tx) => {
      // Insert ledger entry
      await tx.insert(ledgerEntries)
        .values({
          accountId,
          entryType: 'DEBIT',
          amount: debitAmount.toString(),
          referenceType: 'PAYMENT',
          referenceId: 'test-payment-id',
          balanceBefore: balanceBefore.toString(),
          balanceAfter: balanceAfter.toString(),
        });
      
      // Update account balance
      await tx.update(ledgerAccounts)
        .set({ 
          balance: balanceAfter.toString(),
          version: sql`version + 1`,
        })
        .where(eq(ledgerAccounts.id, accountId));
    });
    
    // Verify
    const account = await db.query.ledgerAccounts.findFirst({
      where: eq(ledgerAccounts.id, accountId),
    });
    
    expect(account!.balance).toBe('950.00');
    
    const entries = await db.query.ledgerEntries.findMany({
      where: eq(ledgerEntries.accountId, accountId),
    });
    
    expect(entries.length).toBe(1);
    expect(entries[0]!.entryType).toBe('DEBIT');
    expect(entries[0]!.amount).toBe('50');
  });
  
  it('should maintain double-entry integrity', async () => {
    // Create a second account for double-entry test
    const [settlementAccount] = await db.insert(ledgerAccounts)
      .values({
        userId: testUserId,
        accountType: 'SETTLEMENT',
        currency: 'USDC',
        balance: '0',
      })
      .returning();
    
    const transferAmount = 100.00;
    
    await db.transaction(async (tx) => {
      const sourceAccount = await tx.query.ledgerAccounts.findFirst({
        where: eq(ledgerAccounts.id, accountId),
      });
      
      const sourceBalance = parseFloat(sourceAccount!.balance);
      
      // Debit source
      await tx.insert(ledgerEntries)
        .values({
          accountId,
          entryType: 'DEBIT',
          amount: transferAmount.toString(),
          referenceType: 'TRANSFER',
          referenceId: 'test-transfer-id',
          balanceBefore: sourceBalance.toString(),
          balanceAfter: (sourceBalance - transferAmount).toString(),
        });
      
      await tx.update(ledgerAccounts)
        .set({ balance: (sourceBalance - transferAmount).toString() })
        .where(eq(ledgerAccounts.id, accountId));
      
      // Credit destination
      const destAccount = await tx.query.ledgerAccounts.findFirst({
        where: eq(ledgerAccounts.id, settlementAccount!.id),
      });
      
      const destBalance = parseFloat(destAccount!.balance);
      
      await tx.insert(ledgerEntries)
        .values({
          accountId: settlementAccount!.id,
          entryType: 'CREDIT',
          amount: transferAmount.toString(),
          referenceType: 'TRANSFER',
          referenceId: 'test-transfer-id',
          balanceBefore: destBalance.toString(),
          balanceAfter: (destBalance + transferAmount).toString(),
        });
      
      await tx.update(ledgerAccounts)
        .set({ balance: (destBalance + transferAmount).toString() })
        .where(eq(ledgerAccounts.id, settlementAccount!.id));
    });
    
    // Verify both sides
    const source = await db.query.ledgerAccounts.findFirst({
      where: eq(ledgerAccounts.id, accountId),
    });
    
    const dest = await db.query.ledgerAccounts.findFirst({
      where: eq(ledgerAccounts.id, settlementAccount!.id),
    });
    
    expect(source!.balance).toBe('850.00');
    expect(dest!.balance).toBe('100.00');
  });
});
